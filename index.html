// React 19 + Canvas 2D Interactive Render Loop
export function useCanvasRender(canvasRef, isSpicyMode) {
  useEffect(() => {
    const canvas = canvasRef.current;
    if (!canvas) return;
    const ctx = canvas.getContext('2d', { alpha: true });
    let animationFrameId;

    const render = (timestamp) => {
      ctx.clearRect(0, 0, canvas.width, canvas.height);
      // GPU composited frame rendering with dynamic particle heat aura
      drawFlameParticles(ctx, timestamp, isSpicyMode);
      animationFrameId = requestAnimationFrame(render);
    };
    
    animationFrameId = requestAnimationFrame(render);
    return () => cancelAnimationFrame(animationFrameId);
  }, [isSpicyMode]);
}

// Progressive Video & Image Lazy-Loading Controller
export function initLazyMediaStream() {
  const observer = new IntersectionObserver((entries, obs) => {
    entries.forEach(entry => {
      if (entry.isIntersecting) {
        const media = entry.target;
        if (media.dataset.src) {
          media.src = media.dataset.src;
          media.classList.remove('lazy-blur');
        }
        obs.unobserve(media);
      }
    });
  }, { rootMargin: '200px 0px' });

  document.querySelectorAll('.lazy-media').forEach(el => observer.observe(el));
}

export function initMenuCategoryFilter() {
  const categoryBtns = document.querySelectorAll('.menu-tab-btn');
  const menuItems = document.querySelectorAll('.menu-item-card');

  categoryBtns.forEach(btn => {
    btn.addEventListener('click', () => {
      const cat = btn.dataset.category;
      menuItems.forEach(item => {
        const isMatch = cat === 'ALL' || item.dataset.category === cat;
        item.style.display = isMatch ? 'flex' : 'none';
      });
    });
  });
}

// Client-Side Zero-Reflow Search & Filter Pipeline
export function filterListings(properties, query, activeCategory) {
  const normalized = query.trim().toLowerCase();
  
  // Use batch processing via requestAnimationFrame to prevent DOM thrashing
  requestAnimationFrame(() => {
    properties.forEach(item => {
      const matchesSearch = !normalized || item.dataset.title.toLowerCase().includes(normalized);
      const matchesCat = activeCategory === 'ALL' || item.dataset.category === activeCategory;
      
      const shouldShow = matchesSearch && matchesCat;
      item.style.display = shouldShow ? 'flex' : 'none';
      item.classList.toggle('visible-node', shouldShow);
    });
  });
}
